1//! `MatcherTree` — map-based matching with O(1) exact and O(k) prefix lookup.
2//!
3//! Implements the xDS `MatcherTree` semantics: extract a key via `DataInput`,
4//! then look it up in either an exact map (hash) or a prefix map (radix tree).
5//!
6//! # No fallback lives here
7//!
8//! The proto `MatcherTree` has exactly two members, `input` and the `tree_type`
9//! oneof. `on_no_match` belongs to the enclosing `Matcher`. This type carried
10//! its own until 2026-08-18, which was harmless only because nothing could
11//! construct one from config; wiring it up would have put two fallbacks on one
12//! condition with no stated precedence. See `DECISIONS.md` D-044.
13
14use crate::{radix_tree::RadixTree, DataInput, MatcherError, OnMatch};
15use std::collections::HashMap;
16use std::fmt::Debug;

18/// Tree-based matcher using map lookups instead of predicate evaluation.
19///
20/// This is the xDS `MatcherTree` pattern: extract a key from the context, then
21/// look it up in a map. Cheaper than a linear predicate scan when routing on a
22/// single key, and the only way to express longest-prefix-wins — a
23/// `MatcherList` is first-match-wins in written order, so it returns `/api` for
24/// `/api/v2` whenever `/api` is listed first.
25///
26/// # Variants
27///
28/// - `ExactMatch` — O(1) hash map lookup
29/// - `PrefixMatch` — O(k) radix tree lookup, longest prefix wins
30pub enum MatcherTree<Ctx, A: Clone + Send + Sync + 'static> {
31    /// O(1) exact string lookup.
32    ExactMatch {
33        /// Extracts the lookup key from context.
34        input: Box<dyn DataInput<Ctx>>,
35        /// Map from exact key to outcome.
36        map: HashMap<String, OnMatch<Ctx, A>>,
37    },
38
39    /// O(k) prefix lookup, longest matching prefix wins.
40    PrefixMatch {
41        /// Extracts the lookup key from context.
42        input: Box<dyn DataInput<Ctx>>,
43        /// Radix tree mapping prefixes to outcomes.
44        tree: RadixTree<OnMatch<Ctx, A>>,
45    },
46}
47
48impl<Ctx, A: Clone + Send + Sync + 'static> MatcherTree<Ctx, A> {
49    /// Create an exact-match tree. O(1) hash lookup; keys must match exactly.
50    ///
51    /// # Errors
52    ///
53    /// [`MatcherError::DuplicateTreeKey`] if two entries share a key, and
54    /// [`MatcherError::IncompatibleTypes`] if the input cannot produce a
55    /// string — a tree keyed on an `Int` input can never match anything, so it
56    /// is rejected here rather than silently never firing.
57    pub fn exact<K, I>(input: Box<dyn DataInput<Ctx>>, entries: I) -> Result<Self, MatcherError>
58    where
59        K: Into<String>,
60        I: IntoIterator<Item = (K, OnMatch<Ctx, A>)>,
61    {
62        Self::require_string_input(input.as_ref())?;
63
64        let mut map = HashMap::new();
65        for (key, on_match) in entries {
66            let key = key.into();
67            if map.insert(key.clone(), on_match).is_some() {
68                return Err(MatcherError::DuplicateTreeKey { key });
69            }
70        }
71
72        Ok(Self::ExactMatch { input, map })
73    }
74
75    /// Create a prefix-match tree. O(k) radix lookup, longest prefix wins.
76    ///
77    /// # Errors
78    ///
79    /// As [`exact`](Self::exact).
80    pub fn prefix<K, I>(input: Box<dyn DataInput<Ctx>>, entries: I) -> Result<Self, MatcherError>
81    where
82        K: Into<String>,
83        I: IntoIterator<Item = (K, OnMatch<Ctx, A>)>,
84    {
85        Self::require_string_input(input.as_ref())?;
86
87        let mut tree = RadixTree::new();
88        for (key, on_match) in entries {
89            let key = key.into();
90            // `insert` returns the displaced value. Discarding it is how a
91            // second rule for the same prefix used to disappear silently.
92            if tree.insert(&key, on_match).is_some() {
93                return Err(MatcherError::DuplicateTreeKey { key });
94            }
95        }
96
97        Ok(Self::PrefixMatch { input, tree })
98    }
99
100    /// A tree looks its key up as a string, so a non-string input never matches.
101    fn require_string_input(input: &dyn DataInput<Ctx>) -> Result<(), MatcherError> {
102        if input.data_type() == "string" {
103            return Ok(());
104        }
105        Err(MatcherError::IncompatibleTypes {
106            input_type: input.data_type().to_string(),
107            matcher_types: vec!["string".to_string()],
108        })
109    }
110
111    /// The lookup key for this context, if the input produced a usable string.
112    fn key(&self, ctx: &Ctx) -> Option<String> {
113        let input = match self {
114            Self::ExactMatch { input, .. } | Self::PrefixMatch { input, .. } => input,
115        };
116        input.get(ctx).as_str().map(ToOwned::to_owned)
117    }
118
119    /// The entry a key selects, and which key won.
120    ///
121    /// Split out from [`evaluate`](Self::evaluate) so the traced path can reuse
122    /// the same decision rather than reimplement it — that is where the two
123    /// would otherwise drift apart.
124    fn lookup<'a>(&'a self, key: &'a str) -> Option<(&'a str, &'a OnMatch<Ctx, A>)> {
125        match self {
126            Self::ExactMatch { map, .. } => map
127                .get_key_value(key)
128                .map(|(k, on_match)| (k.as_str(), on_match)),
129            Self::PrefixMatch { tree, .. } => tree.find_longest_prefix_entry(key),
130        }
131    }
132
133    /// Evaluate the tree against a context.
134    ///
135    /// Returns `None` both when no key is available and when the key selects no
136    /// entry. The enclosing [`Matcher`](crate::Matcher) owns the fallback, so
137    /// there is exactly one place a miss can be handled.
138    pub fn evaluate(&self, ctx: &Ctx) -> Option<A> {
139        let key = self.key(ctx)?;
140        let (_, on_match) = self.lookup(&key)?;
141        on_match.evaluate(ctx)
142    }
143
144    /// Which rule this tree applies, and what feeds it — for traces.
145    pub(crate) fn trace_identity(&self) -> (crate::TreeKind, String) {
146        match self {
147            Self::ExactMatch { input, .. } => (crate::TreeKind::Exact, format!("{input:?}")),
148            Self::PrefixMatch { input, .. } => (crate::TreeKind::Prefix, format!("{input:?}")),
149        }
150    }
151
152    pub(crate) fn trace_key(&self, ctx: &Ctx) -> Option<String> {
153        self.key(ctx)
154    }
155
156    pub(crate) fn trace_lookup<'a>(
157        &'a self,
158        key: &'a str,
159    ) -> Option<(&'a str, &'a OnMatch<Ctx, A>)> {
160        self.lookup(key)
161    }
162
163    /// Number of configured entries.
164    pub fn len(&self) -> usize {
165        match self {
166            Self::ExactMatch { map, .. } => map.len(),
167            Self::PrefixMatch { tree, .. } => tree.len(),
168        }
169    }
170
171    /// Returns `true` if the tree has no entries.
172    pub fn is_empty(&self) -> bool {
173        self.len() == 0
174    }
175
176    /// Every outcome this tree can dispatch to.
177    fn outcomes(&self) -> Box<dyn Iterator<Item = &OnMatch<Ctx, A>> + '_> {
178        match self {
179            Self::ExactMatch { map, .. } => Box::new(map.values()),
180            Self::PrefixMatch { tree, .. } => Box::new(tree.values()),
181        }
182    }
183
184    /// Deepest nesting reachable through this tree's entries.
185    ///
186    /// A tree entry can hold a nested `Matcher`, which can hold another tree.
187    /// While this did not exist, `Matcher::depth()` walked past tree entries
188    /// entirely and such a config reported depth 1 — see `DECISIONS.md` D-045.
189    pub(crate) fn depth(&self) -> usize {
190        self.outcomes()
191            .map(|om| match om {
192                OnMatch::Action(_) => 0,
193                OnMatch::Matcher(m) => m.depth(),
194            })
195            .max()
196            .unwrap_or(0)
197    }
198
199    /// Check this tree's width, and recurse into nested matchers.
200    pub(crate) fn validate_widths(&self) -> Result<(), MatcherError> {
201        if self.len() > crate::MAX_TREE_ENTRIES {
202            return Err(MatcherError::TooManyTreeEntries {
203                count: self.len(),
204                max: crate::MAX_TREE_ENTRIES,
205            });
206        }
207
208        for om in self.outcomes() {
209            if let OnMatch::Matcher(nested) = om {
210                nested.validate_widths()?;
211            }
212        }
213
214        Ok(())
215    }
216}
217
218impl<Ctx, A: Clone + Send + Sync + Debug + 'static> Debug for MatcherTree<Ctx, A> {
219    fn fmt(&self, f: &mut std::fmt::Formatter<'_>) -> std::fmt::Result {
220        match self {
221            Self::ExactMatch { input, map } => f
222                .debug_struct("ExactMatch")
223                .field("input", input)
224                .field("entries", &map.len())
225                .finish(),
226            Self::PrefixMatch { input, tree } => f
227                .debug_struct("PrefixMatch")
228                .field("input", input)
229                .field("tree", tree)
230                .finish(),
231        }
232    }
233}

255        MatcherTree::exact(
256            Box::new(PathInput),
257            entries

278    #[test]
279    fn exact_lookup_hits_its_own_entry() {
280        let tree = exact(vec![("/health", "health"), ("/ready", "ready")]);
281
282        assert_eq!(tree.evaluate(&ctx("/health")), Some("health".into()));
283        assert_eq!(tree.evaluate(&ctx("/ready")), Some("ready".into()));
284    }

286    #[test]
287    fn exact_means_exact() {
288        let tree = exact(vec![("/a", "a")]);
289
290        // A miss is `None`. The fallback belongs to the enclosing `Matcher`,
291        // so there is nothing here to fall back to — see D-044.
292        assert_eq!(tree.evaluate(&ctx("/b")), None);
293        assert_eq!(tree.evaluate(&ctx("/ab")), None);
294    }

296    #[test]
297    fn prefix_lookup_takes_the_longest_match() {
298        let tree = prefix(vec![("/", "root"), ("/api", "api"), ("/api/v2", "api_v2")]);
299
300        assert_eq!(tree.evaluate(&ctx("/api/v2/users")), Some("api_v2".into()));
301        assert_eq!(tree.evaluate(&ctx("/api/v1/users")), Some("api".into()));
302        assert_eq!(tree.evaluate(&ctx("/other")), Some("root".into()));
303        assert_eq!(tree.evaluate(&ctx("nope")), None);
304    }

368    #[test]
369    fn an_input_that_lies_about_its_type_still_silently_misses() {
370        // This is finding F17, not a gap in the check above. `data_type()`
371        // defaults to `"string"` (`data_input.rs`), so an input that returns
372        // `Int` without overriding it *declares* itself a string. Construction
373        // cannot catch that, and the lookup then finds nothing.
374        //
375        // The test exists to pin the residue: the construction check is worth
376        // having, and it is bounded by the default. Closing F17 is what would
377        // make this case impossible, and that is not this change.
378        #[derive(Debug)]
379        struct UndeclaredIntInput;
380
381        impl DataInput<TestContext> for UndeclaredIntInput {
382            fn get(&self, _ctx: &TestContext) -> MatchingData {
383                MatchingData::Int(42)
384            }
385        }
386
387        let tree = MatcherTree::<TestContext, String>::exact(
388            Box::new(UndeclaredIntInput),
389            [("42", OnMatch::Action("matched".to_string()))],
390        )
391        .expect("declares itself a string, so it loads");
392
393        assert_eq!(tree.evaluate(&ctx("ignored")), None);
394    }
