1//! Action sites numbered in document order, so a binding returns an index
2//! and holds the actions itself (D-056).
3//!
4//! The walk is the same for every consumer of this crate, which is what
5//! makes the PyO3 and wasm bindings number a document identically: a
6//! pre-order walk of the IR, list entries in written order, tree entries by
7//! sorted key, and each matcher's fallback after its entries. Two loads of
8//! one document produce one table.
9
10use rumi::{
11    FieldMatcherIr, MatcherIr, MatcherKindIr, MatcherTreeIr, OnMatchIr, TreeTypeIr,
12    TypedExtensionConfig,
13};
14
15/// The position of an action site in the document's action table.
16///
17/// The engine's `A` for a binding: what `evaluate` returns across the FFI,
18/// with the action itself looked up host-side (D-056).
19#[derive(Debug, Clone, Copy, PartialEq, Eq, Hash, PartialOrd, Ord)]
20pub struct ActionIndex(pub u32);
21
22impl std::fmt::Display for ActionIndex {
23    fn fmt(&self, f: &mut std::fmt::Formatter<'_>) -> std::fmt::Result {
24        write!(f, "#{}", self.0)
25    }
26}
27
28/// A document with its action sites numbered.
29#[derive(Debug, Clone)]
30pub struct Indexed {
31    /// The IR with every action replaced by its index.
32    pub ir: MatcherIr<ActionIndex>,
33    /// The actions, in index order: `table[i.0]` is what `ActionIndex(i)`
34    /// stands for. One entry per site, so the mapping is a bijection over the
35    /// document and identical for every load of it.
36    pub table: Vec<TypedExtensionConfig>,
37    /// The first index under the root matcher's `on_no_match`, if it has one.
38    /// The root fallback is walked last, so every index from here on is a
39    /// fallback decision (D-056's `fallback` flag).
40    pub fallback_start: Option<ActionIndex>,
41}
42
43/// Replace every action in the IR with its index, returning the actions in
44/// index order.
45#[must_use]
46pub fn index_actions(ir: MatcherIr<TypedExtensionConfig>) -> Indexed {
47    let mut table = Vec::new();
48    let kind = walk_kind(ir.kind, &mut table);
49    let fallback_start = ir
50        .on_no_match
51        .as_ref()
52        .map(|_| ActionIndex(u32::try_from(table.len()).expect("fewer than 2^32 action sites")));
53    let on_no_match = ir.on_no_match.map(|om| walk_on_match(om, &mut table));
54    Indexed {
55        ir: MatcherIr { kind, on_no_match },
56        table,
57        fallback_start,
58    }
59}
60
61fn walk_matcher(
62    ir: MatcherIr<TypedExtensionConfig>,
63    table: &mut Vec<TypedExtensionConfig>,
64) -> MatcherIr<ActionIndex> {
65    let kind = walk_kind(ir.kind, table);
66    let on_no_match = ir.on_no_match.map(|om| walk_on_match(om, table));
67    MatcherIr { kind, on_no_match }
68}
69
70fn walk_kind(
71    kind: MatcherKindIr<TypedExtensionConfig>,
72    table: &mut Vec<TypedExtensionConfig>,
73) -> MatcherKindIr<ActionIndex> {
74    match kind {
75        MatcherKindIr::List(matchers) => MatcherKindIr::List(
76            matchers
77                .into_iter()
78                .map(|fm| FieldMatcherIr {
79                    predicate: fm.predicate,
80                    on_match: walk_on_match(fm.on_match, table),
81                })
82                .collect(),
83        ),
84        MatcherKindIr::Tree(tree) => MatcherKindIr::Tree(MatcherTreeIr {
85            input: tree.input,
86            tree: match tree.tree {
87                TreeTypeIr::ExactMatchMap(entries) => {
88                    TreeTypeIr::ExactMatchMap(walk_entries(entries, table))
89                }
90                TreeTypeIr::PrefixMatchMap(entries) => {
91                    TreeTypeIr::PrefixMatchMap(walk_entries(entries, table))
92                }
93            },
94        }),
95    }
96}
97
98fn walk_entries(
99    mut entries: Vec<(String, OnMatchIr<TypedExtensionConfig>)>,
100    table: &mut Vec<TypedExtensionConfig>,
101) -> Vec<(String, OnMatchIr<ActionIndex>)> {
102    entries.sort_by(|a, b| a.0.cmp(&b.0));
103    entries
104        .into_iter()
105        .map(|(key, om)| (key, walk_on_match(om, table)))
106        .collect()
107}
108
109fn walk_on_match(
110    om: OnMatchIr<TypedExtensionConfig>,
111    table: &mut Vec<TypedExtensionConfig>,
112) -> OnMatchIr<ActionIndex> {
113    match om {
114        OnMatchIr::Action { action } => {
115            let index = u32::try_from(table.len()).expect("fewer than 2^32 action sites");
116            table.push(action);
117            OnMatchIr::Action {
118                action: ActionIndex(index),
119            }
120        }
121        OnMatchIr::Matcher { matcher } => OnMatchIr::Matcher {
122            matcher: Box::new(walk_matcher(*matcher, table)),
123        },
124    }
125}

127#[cfg(test)]
128mod tests {
129    use super::*;
130    use rumi::{PredicateIr, SinglePredicateIr, StringMatchSpec, ValueMatchIr};
131
132    fn named(name: &str) -> TypedExtensionConfig {
133        TypedExtensionConfig {
134            type_url: "xuma.core.v1.NamedAction".into(),
135            config: serde_json::json!({ "name": name }),
136        }
137    }
138
139    fn action(name: &str) -> OnMatchIr<TypedExtensionConfig> {
140        OnMatchIr::Action {
141            action: named(name),
142        }
143    }
144
145    fn rule(
146        name: &str,
147        on_match: OnMatchIr<TypedExtensionConfig>,
148    ) -> FieldMatcherIr<TypedExtensionConfig> {
149        FieldMatcherIr {
150            predicate: PredicateIr::Single(SinglePredicateIr {
151                input: TypedExtensionConfig {
152                    type_url: "xuma.core.v1.MapInput".into(),
153                    config: serde_json::json!({ "key": name }),
154                },
155                matcher: ValueMatchIr::BuiltIn {
156                    spec: StringMatchSpec::Exact(name.into()),
157                    ignore_case: false,
158                },
159            }),
160            on_match,
161        }
162    }
163
164    fn names(table: &[TypedExtensionConfig]) -> Vec<String> {
165        table
166            .iter()
167            .map(|t| t.config["name"].as_str().unwrap().to_owned())
168            .collect()
169    }
170
171    fn actions_of(ir: &MatcherIr<ActionIndex>) -> Vec<u32> {
172        fn on_match(om: &OnMatchIr<ActionIndex>, out: &mut Vec<u32>) {
173            match om {
174                OnMatchIr::Action { action } => out.push(action.0),
175                OnMatchIr::Matcher { matcher } => matcher_(matcher, out),
176            }
177        }
178        fn matcher_(ir: &MatcherIr<ActionIndex>, out: &mut Vec<u32>) {
179            match &ir.kind {
180                MatcherKindIr::List(ms) => ms.iter().for_each(|fm| on_match(&fm.on_match, out)),
181                MatcherKindIr::Tree(t) => match &t.tree {
182                    TreeTypeIr::ExactMatchMap(e) | TreeTypeIr::PrefixMatchMap(e) => {
183                        e.iter().for_each(|(_, om)| on_match(om, out));
184                    }
185                },
186            }
187            if let Some(om) = &ir.on_no_match {
188                on_match(om, out);
189            }
190        }
191        let mut out = Vec::new();
192        matcher_(ir, &mut out);
193        out
194    }

196    #[test]
197    fn a_list_is_numbered_in_written_order_with_the_fallback_last() {
198        let ir = MatcherIr::list(vec![
199            rule("a", action("first")),
200            rule("b", action("second")),
201        ])
202        .with_fallback(Some(action("fallback")));
203
204        let Indexed {
205            ir,
206            table,
207            fallback_start,
208        } = index_actions(ir);
209
210        assert_eq!(names(&table), ["first", "second", "fallback"]);
211        assert_eq!(actions_of(&ir), [0, 1, 2]);
212        assert_eq!(fallback_start, Some(ActionIndex(2)));
213    }

215    #[test]
216    fn a_nested_matcher_is_numbered_where_it_appears() {
217        let nested = MatcherIr::list(vec![rule("x", action("inner"))])
218            .with_fallback(Some(action("inner_fallback")));
219        let ir = MatcherIr::list(vec![
220            rule("a", action("before")),
221            rule(
222                "b",
223                OnMatchIr::Matcher {
224                    matcher: Box::new(nested),
225                },
226            ),
227            rule("c", action("after")),
228        ]);
229
230        let indexed = index_actions(ir);
231
232        assert_eq!(
233            names(&indexed.table),
234            ["before", "inner", "inner_fallback", "after"]
235        );
236        assert_eq!(
237            indexed.fallback_start, None,
238            "a nested fallback is not the root's"
239        );
240    }

242    #[test]
243    fn tree_entries_are_numbered_by_sorted_key_whatever_order_they_arrived_in() {
244        let tree = |entries: Vec<(&str, &str)>| MatcherIr {
245            kind: MatcherKindIr::Tree(MatcherTreeIr {
246                input: TypedExtensionConfig {
247                    type_url: "xuma.core.v1.MapInput".into(),
248                    config: serde_json::json!({ "key": "role" }),
249                },
250                tree: TreeTypeIr::ExactMatchMap(
251                    entries
252                        .into_iter()
253                        .map(|(k, a)| (k.to_owned(), action(a)))
254                        .collect(),
255                ),
256            }),
257            on_no_match: Some(action("deny")),
258        };
259
260        let one = index_actions(tree(vec![("viewer", "read"), ("admin", "allow")])).table;
261        let two = index_actions(tree(vec![("admin", "allow"), ("viewer", "read")])).table;
262
263        assert_eq!(names(&one), ["allow", "read", "deny"]);
264        assert_eq!(names(&one), names(&two));
265    }

267    #[test]
268    fn every_index_names_exactly_one_site() {
269        let ir = MatcherIr::list(vec![rule("a", action("same")), rule("b", action("same"))]);

271        let indexed = index_actions(ir);
272
273        assert_eq!(indexed.table.len(), 2, "two sites, even with one name");
