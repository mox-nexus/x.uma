1//! Human-readable rendering of an `EvalTrace`.
2//!
3//! Answers the question people actually ask, which is not "what matched" but
4//! "why did *that* one match and not mine". So the output leads with the rule
5//! index, shows the value that was extracted, and marks the winner.
6
7use std::fmt::Display;
8
9use rumi::{
10    EvalSteps, EvalTrace, MatchResult, OnMatchTrace, PredicateTrace, TreeKind, TreeLookupTrace,
11};
12
13/// Render a trace to stdout.
14pub fn print<A: Display>(trace: &EvalTrace<A>) {
15    match &trace.steps {
16        EvalSteps::List(steps) => {
17            if steps.is_empty() {
18                println!("no rules were evaluated (empty matcher list)");
19            }
20            for step in steps {
21                let mark = if step.matched { "MATCH" } else { "  -  " };
22                println!("[{mark}] rule {}", step.index);
23                print_predicate(&step.predicate_trace, 2);
24            }
25        }
26        EvalSteps::Tree(lookup) => print_tree_lookup(lookup, trace.undecided),
27    }
28
29    println!();
30    match &trace.result {
31        None if trace.undecided => {
32            println!("=> (undecided: a required value is absent; no rule and no on_no_match)");
33        }
34        Some(action) if trace.used_fallback => {
35            println!("=> {action}   (no rule matched; on_no_match)");
36        }
37        Some(action) => println!("=> {action}"),
38        None => println!("=> (no match, and no on_no_match)"),
39    }
40}

42/// A tree does one lookup, so the trace reports the lookup rather than a list
43/// of rules. The three ways it can miss are kept distinct: no usable key at
44/// all, a key that matched nothing, and a key that matched an entry whose
45/// nested matcher then failed. Collapsing them would send a config author
46/// looking in the wrong place.
47fn print_tree_lookup<A: Display>(t: &TreeLookupTrace<A>, undecided: bool) {
48    let rule = match t.kind {
49        TreeKind::Exact => "exact",
50        TreeKind::Prefix => "longest-prefix",
51    };
52    println!("[TREE ] {rule} lookup on {}", t.input);
53
54    match &t.key {
55        None if undecided => println!("  key: (required, and absent — undecided)"),
56        None => println!("  key: (input produced no string — nothing to look up)"),
57        Some(key) if t.attempts.is_empty() => println!("  key: {key:?} — no entry"),
58        Some(key) => println!("  key: {key:?}"),
59    }
60    for attempt in &t.attempts {
61        let entry = &attempt.matched_key;
62        match &attempt.on_match {
63            OnMatchTrace::Action(_) => println!("  entry {entry:?} — matched"),
64            OnMatchTrace::Nested(nested) if nested.result.is_some() => {
65                println!("  entry {entry:?} — matched, nested matcher decided");
66            }
67            OnMatchTrace::Nested(_) => {
68                println!(
69                    "  entry {entry:?} — nested matcher missed, trying the next shorter prefix"
70                );
71            }
72        }
73    }
74}

76fn print_predicate(t: &PredicateTrace, indent: usize) {
77    let pad = " ".repeat(indent);
78    match t {
79        PredicateTrace::Single {
80            result,
81            input,
82            data,
83            matcher,
84        } => {
85            // `data` is the value the input actually pulled out of the context.
86            // Showing it is the whole point: a rule that never fires usually
87            // means the input read nothing.
88            println!("{pad}{} {input} = {data}  vs  {matcher}", mark(*result));
89        }
90        PredicateTrace::And { result, children } => {
91            println!("{pad}{} AND", mark(*result));
92            for c in children {
93                print_predicate(c, indent + 2);
94            }
95        }
96        PredicateTrace::Or { result, children } => {
97            println!("{pad}{} OR", mark(*result));
98            for c in children {
99                print_predicate(c, indent + 2);
100            }
101        }
102        PredicateTrace::Not { result, inner } => {
103            println!("{pad}{} NOT", mark(*result));
104            print_predicate(inner, indent + 2);
105        }
106    }
107}

109fn mark(result: MatchResult) -> &'static str {
110    match result {
111        MatchResult::Matched => "✓",
112        MatchResult::NoMatch => "✗",
113        MatchResult::Undecided => "?",
114    }
