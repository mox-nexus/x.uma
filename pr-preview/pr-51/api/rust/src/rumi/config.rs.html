1//! Config types for generic matcher construction.
2//!
3//! These mirror the runtime matcher types and are what
4//! [`Registry::load_matcher()`] consumes. They are **built, not parsed**: the
5//! authored config format is canonical protojson, read by `rumi-proto` into
6//! xDS proto types and converted here.
7//!
8//! None of the tree-shape types below derives `Deserialize`, and that is the
9//! enforcement rather than a style choice. While they did, a second, terser
10//! JSON dialect remained authorable in Rust — one no fixture used, no other
11//! implementation could read, and nothing documented. Removing the impl makes
12//! writing that dialect a compile error instead of a convention.
13//!
14//! The payload types keep theirs: a `TypedConfig.config` body is deserialized
15//! as the registered factory's `Config` associated type, which is the
16//! extension seam working as designed.
17//!
18//! # Relationship to runtime types
19//!
20//! | Config type | Runtime type | Loader method |
21//! |-------------|-------------|---------------|
22//! | [`MatcherConfig`] | [`Matcher`](crate::Matcher) | `Registry::load_matcher()` |
23//! | [`FieldMatcherConfig`] | [`FieldMatcher`](crate::FieldMatcher) | `Registry::load_field_matcher()` |
24//! | [`PredicateConfig`] | [`Predicate`](crate::Predicate) | `Registry::load_predicate()` |
25//! | [`SinglePredicateConfig`] | [`SinglePredicate`](crate::SinglePredicate) | `Registry::load_single()` |
26//! | [`ValueMatchConfig`] | `Box<dyn InputMatcher>` | built-in or via registry factory |
27//! | [`OnMatchConfig`] | [`OnMatch`](crate::OnMatch) | `Registry::load_on_match()` |
28//! | [`TypedConfig`] | `Box<dyn DataInput<Ctx>>` or `Box<dyn InputMatcher>` | via registry factory |

33/// Configuration for a [`Matcher`](crate::Matcher).
34///
35/// Built by a protojson reader, then loaded into a runtime `Matcher` via
36/// [`Registry::load_matcher()`](crate::Registry::load_matcher).
37#[derive(Debug, Clone)]
38pub struct MatcherConfig<A> {
39    /// Field matchers to evaluate in order (first-match-wins).
40    pub matchers: Vec<FieldMatcherConfig<A>>,
41
42    /// Fallback when no field matcher matches.
43    pub on_no_match: Option<OnMatchConfig<A>>,
44}
45
46/// Configuration for a [`FieldMatcher`](crate::FieldMatcher).
47#[derive(Debug, Clone)]
48pub struct FieldMatcherConfig<A> {
49    /// The predicate that gates this field matcher.
50    pub predicate: PredicateConfig,
51
52    /// What to do when the predicate matches.
53    pub on_match: OnMatchConfig<A>,
54}

56/// Configuration for a [`Predicate`](crate::Predicate).
57///
58/// Mirrors xDS `Matcher.MatcherList.Predicate`, whose `oneof match_type` is
59/// these four arms.
60#[derive(Debug, Clone)]
61pub enum PredicateConfig {
62    /// A single predicate: input + value match.
63    Single(SinglePredicateConfig),
64
65    /// All predicates must match (logical AND).
66    And {
67        /// Child predicates (all must match).
68        predicates: Vec<PredicateConfig>,
69    },
70
71    /// Any predicate must match (logical OR).
72    Or {
73        /// Child predicates (any must match).
74        predicates: Vec<PredicateConfig>,
75    },
76
77    /// Inverts the inner predicate (logical NOT).
78    Not {
79        /// The predicate to negate.
80        predicate: Box<PredicateConfig>,
81    },
82}
83
84/// How to match the extracted value in a [`SinglePredicateConfig`].
85///
86/// Mirrors Envoy's `oneof matcher` in `SinglePredicate`:
87/// - `BuiltIn` — built-in string matching (Envoy: `StringMatcher value_match`)
88/// - `Custom` — custom matcher via registry (Envoy: `TypedExtensionConfig custom_match`)
89///
90/// The enum makes illegal states unrepresentable: exactly one variant is active.
91#[derive(Debug, Clone)]
92pub enum ValueMatchConfig {
93    /// Built-in string matching (exact, prefix, suffix, contains, regex).
94    BuiltIn {
95        /// The pattern and how to apply it.
96        spec: StringMatchSpec,
97        /// Case-insensitive matching — xDS `StringMatcher.ignore_case`.
98        ///
99        /// Carried here rather than inside `StringMatchSpec` because it is a
100        /// property of the comparison, not of the pattern. The proto path used
101        /// to drop it on the floor.
102        ignore_case: bool,
103    },
104    /// Custom matcher resolved via the registry's matcher factories.
105    Custom(TypedConfig),
106}

108/// Configuration for a [`SinglePredicate`](crate::SinglePredicate).
109///
110/// Combines a typed input reference (resolved via registry) with a value
111/// matcher. Mirrors xDS `SinglePredicate`, whose `oneof matcher` is either a
112/// built-in `StringMatcher` or a `TypedExtensionConfig`.
113///
114/// The exactly-one-of rule is enforced where the config is built: the proto
115/// `oneof` makes both-set unrepresentable, and `rumi-proto`'s converter
116/// rejects neither-set.
117#[derive(Debug, Clone)]
118pub struct SinglePredicateConfig {
119    /// The input to extract data from context.
120    /// Resolved at load time via the registry's `type_url` lookup.
121    pub input: TypedConfig,
122
123    /// How to match the extracted value.
124    pub matcher: ValueMatchConfig,
125}
126
127/// Reference to a registered type with its configuration.
128///
129/// Maps to xDS `TypedExtensionConfig`:
130/// - `type_url` identifies the registered type (input, matcher, or action)
131/// - `config` carries the type-specific configuration payload
132#[derive(Debug, Clone)]
133pub struct TypedConfig {
134    /// The type URL identifying the registered type.
135    /// Must match a `type_url` registered in the [`Registry`](crate::Registry).
136    pub type_url: String,
137
138    /// Type-specific configuration payload.
139    /// Deserialized as the `Config` associated type of the registered trait impl.
140    pub config: serde_json::Value,
141}
142
143/// Empty configuration for [`DataInput`](crate::DataInput) types that need no configuration.
144///
145/// Accepts any JSON value (`{}`, `null`, etc.) and ignores it.
146/// Use as the `Config` associated type in [`IntoDataInput`](crate::IntoDataInput)
147/// for inputs that are self-contained (no construction parameters).
148#[derive(Debug, Clone, Copy)]
149pub struct UnitConfig;
150
151impl<'de> Deserialize<'de> for UnitConfig {
152    fn deserialize<D: serde::Deserializer<'de>>(deserializer: D) -> Result<Self, D::Error> {
153        serde::de::IgnoredAny::deserialize(deserializer)?;
154        Ok(UnitConfig)
155    }
156}

158/// Configuration for [`OnMatch`](crate::OnMatch).
159///
160/// Either an action (leaf) or a nested matcher (tree).
161/// `OnMatch` exclusivity is enforced by the enum: action XOR matcher.
162#[derive(Debug, Clone)]
163pub enum OnMatchConfig<A> {
164    /// Return this action when the predicate matches.
165    Action {
166        /// The action value.
167        action: A,
168    },
169
170    /// Evaluate a nested matcher when the predicate matches.
171    Matcher {
172        /// The nested matcher configuration.
173        matcher: Box<MatcherConfig<A>>,
174    },
175}
