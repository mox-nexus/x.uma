1//! Compiler: Gateway API `HttpRouteMatch` -> rumi Matcher
2//!
3//! Translates user-friendly Gateway API configuration into efficient
4//! runtime matchers operating on `HttpMessage`.
5
6use crate::inputs::{HeaderInput, MethodInput, PathInput, QueryParamInput};
7use crate::message::HttpMessage;
8use k8s_gateway_api::{HttpHeaderMatch, HttpPathMatch, HttpQueryParamMatch, HttpRouteMatch};
9use rumi::prelude::*;
10use rumi::{MatcherError, StringMatchSpec};
11
12/// Extension trait for compiling `HttpRouteMatch` to rumi Matcher.
13///
14/// Both methods return `Result`. Until 2026-08-17 they did not: an invalid
15/// regex was silently replaced with an exact match on the *pattern literal*,
16/// so a route the operator believed was live simply never fired. The sibling
17/// Claude compiler already returned `Result`; this removes the asymmetry.
18pub trait HttpRouteMatchExt {
19    /// Compile this `HttpRouteMatch` into a rumi Matcher.
20    ///
21    /// The resulting matcher operates on `HttpMessage` and returns
22    /// the provided action when all conditions match.
23    ///
24    /// # Errors
25    ///
26    /// Returns [`MatcherError::InvalidPattern`] if a regex does not compile, or
27    /// [`MatcherError::PatternTooLong`] if one exceeds the configured limits.
28    fn compile<A: Clone + Send + Sync + 'static>(
29        &self,
30        action: A,
31    ) -> Result<Matcher<HttpMessage, A>, MatcherError>;
32
33    /// Compile this `HttpRouteMatch` into a Predicate (without action).
34    ///
35    /// # Errors
36    ///
37    /// As [`compile`](Self::compile).
38    fn to_predicate(&self) -> Result<Predicate<HttpMessage>, MatcherError>;
39}
40
41impl HttpRouteMatchExt for HttpRouteMatch {
42    fn compile<A: Clone + Send + Sync + 'static>(
43        &self,
44        action: A,
45    ) -> Result<Matcher<HttpMessage, A>, MatcherError> {
46        Ok(Matcher::from_predicate(self.to_predicate()?, action, None))
47    }
48
49    fn to_predicate(&self) -> Result<Predicate<HttpMessage>, MatcherError> {
50        let mut predicates: Vec<Predicate<HttpMessage>> = Vec::new();
51
52        // Path matching
53        if let Some(path_match) = &self.path {
54            predicates.push(compile_path_match(path_match)?);
55        }
56
57        // Method matching
58        if let Some(method) = &self.method {
59            predicates.push(Predicate::Single(SinglePredicate::new(
60                Box::new(MethodInput),
61                Box::new(ExactMatcher::new(method.as_str())),
62            )));
63        }
64
65        // Header matching (all headers are ANDed)
66        if let Some(headers) = &self.headers {
67            for header_match in headers {
68                predicates.push(compile_header_match(header_match)?);
69            }
70        }
71
72        // Query param matching (all params are ANDed)
73        if let Some(query_params) = &self.query_params {
74            for query_match in query_params {
75                predicates.push(compile_query_param_match(query_match)?);
76            }
77        }
78
79        Ok(Predicate::from_all(predicates, catch_all()))
80    }
81}
82
83/// A catch-all predicate that matches any HTTP request (empty prefix = match all paths).
84fn catch_all() -> Predicate<HttpMessage> {
85    Predicate::Single(SinglePredicate::new(
86        Box::new(PathInput),
87        Box::new(PrefixMatcher::new("")),
88    ))
89}

91/// Compile a path match to a predicate.
92///
93/// Goes through [`StringMatchSpec::to_input_matcher`], which owns the pattern
94/// length limits. Constructing matchers directly here is what let this compiler
95/// bypass every declared limit.
96fn compile_path_match(path_match: &HttpPathMatch) -> Result<Predicate<HttpMessage>, MatcherError> {
97    let spec = match path_match {
98        HttpPathMatch::Exact { value } => StringMatchSpec::Exact(value.clone()),
99        HttpPathMatch::PathPrefix { value } => StringMatchSpec::Prefix(value.clone()),
100        HttpPathMatch::RegularExpression { value } => StringMatchSpec::Regex(value.clone()),
101    };
102    Ok(Predicate::Single(SinglePredicate::new(
103        Box::new(PathInput),
104        spec.to_input_matcher()?,
105    )))
106}
107
108/// Compile a header match to a predicate.
109fn compile_header_match(
110    header_match: &HttpHeaderMatch,
111) -> Result<Predicate<HttpMessage>, MatcherError> {
112    let (name, spec) = match header_match {
113        HttpHeaderMatch::Exact { name, value } => (name, StringMatchSpec::Exact(value.clone())),
114        HttpHeaderMatch::RegularExpression { name, value } => {
115            (name, StringMatchSpec::Regex(value.clone()))
116        }
117    };
118    Ok(Predicate::Single(SinglePredicate::new(
119        Box::new(HeaderInput::new(name.as_str())),
120        spec.to_input_matcher()?,
121    )))
122}
123
124/// Compile a query param match to a predicate.
125fn compile_query_param_match(
126    query_match: &HttpQueryParamMatch,
127) -> Result<Predicate<HttpMessage>, MatcherError> {
128    let (name, spec) = match query_match {
129        HttpQueryParamMatch::Exact { name, value } => (name, StringMatchSpec::Exact(value.clone())),
130        HttpQueryParamMatch::RegularExpression { name, value } => {
131            (name, StringMatchSpec::Regex(value.clone()))
132        }
133    };
134    Ok(Predicate::Single(SinglePredicate::new(
135        Box::new(QueryParamInput::new(name.as_str())),
136        spec.to_input_matcher()?,
137    )))
138}

140/// Compile multiple `HttpRouteMatch` entries into a single Matcher.
141///
142/// Multiple matches are `ORed` together per Gateway API semantics.
143/// # Errors
144///
145/// Returns [`MatcherError::InvalidPattern`] if any regex does not compile, or
146/// [`MatcherError::PatternTooLong`] if any pattern exceeds the limits.
147pub fn compile_route_matches<A: Clone + Send + Sync + 'static>(
148    matches: &[HttpRouteMatch],
149    action: A,
150    on_no_match: Option<A>,
151) -> Result<Matcher<HttpMessage, A>, MatcherError> {
152    let predicates: Vec<Predicate<HttpMessage>> = matches
153        .iter()
154        .map(HttpRouteMatchExt::to_predicate)
155        .collect::<Result<_, _>>()?;
156
157    Ok(Matcher::from_predicate(
158        Predicate::from_any(predicates, catch_all()),
159        action,
160        on_no_match,
161    ))
162}

175    struct RequestBuilder {
176        headers: Vec<HeaderValue>,
177    }
178
179    impl RequestBuilder {
180        fn new() -> Self {
181            Self { headers: vec![] }
182        }
183
184        fn path(mut self, path: &str) -> Self {
185            self.headers.push(HeaderValue {
186                key: ":path".into(),
187                value: path.into(),
188                raw_value: vec![],
189            });
190            self
191        }
192
193        fn method(mut self, method: &str) -> Self {
194            self.headers.push(HeaderValue {
195                key: ":method".into(),
196                value: method.into(),
197                raw_value: vec![],
198            });
199            self
200        }
201
202        fn header(mut self, name: &str, value: &str) -> Self {
203            self.headers.push(HeaderValue {
204                key: name.to_lowercase(),
205                value: value.into(),
206                raw_value: vec![],
207            });
208            self
209        }
210
211        fn build(self) -> HttpMessage {
212            let req = ProcessingRequest {
213                request: Some(Request::RequestHeaders(HttpHeaders {
214                    headers: Some(HeaderMap {
215                        headers: self.headers,
216                    }),
217                    ..Default::default()
218                })),
219                ..Default::default()
220            };
221            HttpMessage::from(&req)
222        }
223    }
