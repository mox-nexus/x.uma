1//! `StringMatchSpec` — Config-level string match specification
2//!
3//! This type represents a user's *intent* for string matching (e.g., "exact match on /api").
4//! It compiles to runtime [`InputMatcher`] types via [`to_input_matcher()`](StringMatchSpec::to_input_matcher).
5//!
6//! # Naming: Spec vs Matcher
7//!
8//! - [`StringMatchSpec`] = config-level specification (what the user wrote)
9//! - [`StringMatcher`](crate::StringMatcher) = runtime engine (what evaluates at match time)
10//!
11//! The `Spec` suffix makes the ontological distinction clear (Karman: guild review).
12
13use crate::{DataInput, InputMatcher, MatcherError, Predicate, SinglePredicate, StringMatcher};
14use std::fmt;

16/// A string match specification from user configuration.
17///
18/// Represents one of five matching strategies. Compiles to the appropriate
19/// runtime [`InputMatcher`] via [`to_input_matcher()`](Self::to_input_matcher).
20///
21/// # Example
22///
23/// ```
24/// use rumi::StringMatchSpec;
25///
26/// let spec = StringMatchSpec::Prefix("/api".into());
27/// let matcher = spec.to_input_matcher().unwrap();
28///
29/// // Or compile directly to a Predicate with a DataInput:
30/// // let predicate = spec.to_predicate(Box::new(PathInput))?;
31/// ```
32#[derive(Debug, Clone)]
33#[cfg_attr(feature = "serde", derive(serde::Serialize, serde::Deserialize))]
34pub enum StringMatchSpec {
35    /// Exact string equality.
36    Exact(String),
37    /// String starts with prefix.
38    Prefix(String),
39    /// String ends with suffix.
40    Suffix(String),
41    /// String contains substring.
42    Contains(String),
43    /// Regular expression match (Rust `regex` crate syntax, linear time).
44    Regex(String),
45}
46
47impl StringMatchSpec {
48    /// Compile this spec into a runtime [`InputMatcher`].
49    ///
50    /// Pattern length limits are enforced **here**, in the constructor, not in
51    /// the config loader. This is the only path from a spec to a matcher, so
52    /// every caller inherits the guarantee — the registry, both domain
53    /// compilers, and anyone constructing a spec by hand.
54    ///
55    /// They lived in a private `Registry` method until 2026-08-17, which meant
56    /// `HookMatch::compile` accepted an 8 MB pattern against an 8192-byte limit.
57    ///
58    /// # Errors
59    ///
60    /// Returns [`MatcherError::PatternTooLong`] if the pattern exceeds
61    /// [`MAX_PATTERN_LENGTH`](crate::MAX_PATTERN_LENGTH), or
62    /// [`MAX_REGEX_PATTERN_LENGTH`](crate::MAX_REGEX_PATTERN_LENGTH) for a
63    /// regex. Returns [`MatcherError::InvalidPattern`] if the regex is invalid.
64    pub fn to_input_matcher(&self) -> Result<Box<dyn InputMatcher>, MatcherError> {
65        self.to_input_matcher_with_case(false)
66    }
67
68    /// Compile this spec, optionally matching case-insensitively.
69    ///
70    /// This is xDS `StringMatcher.ignore_case`. The flag was read from the
71    /// proto and thrown away until 2026-08-18, under a comment asserting the
72    /// registry handled it — a rule that read case-insensitive and was not.
73    ///
74    /// # Errors
75    ///
76    /// As [`to_input_matcher`](Self::to_input_matcher), plus
77    /// [`MatcherError::InvalidPattern`] when `ignore_case` is set on a regex
78    /// that countermands it inline — see [`disables_case_insensitivity`].
79    pub fn to_input_matcher_with_case(
80        &self,
81        ignore_case: bool,
82    ) -> Result<Box<dyn InputMatcher>, MatcherError> {
83        self.check_length()?;
84
85        let boxed = |sm: StringMatcher| Box::new(sm) as Box<dyn InputMatcher>;
86        match self {
87            Self::Exact(v) => Ok(boxed(StringMatcher::exact(v.clone(), ignore_case))),
88            Self::Prefix(v) => Ok(boxed(StringMatcher::prefix(v.clone(), ignore_case))),
89            Self::Suffix(v) => Ok(boxed(StringMatcher::suffix(v.clone(), ignore_case))),
90            Self::Contains(v) => Ok(boxed(StringMatcher::contains(v.clone(), ignore_case))),
91            Self::Regex(v) => {
92                if ignore_case && disables_case_insensitivity(v) {
93                    return Err(MatcherError::InvalidPattern {
94                        pattern: v.clone(),
95                        source: "ignore_case is set, but the pattern turns case-insensitivity \
96                                 off inline with a (?-i) flag. An inline flag wins, so this \
97                                 rule would read case-insensitive and not be. Remove one of \
98                                 the two."
99                            .into(),
100                    });
101                }
102                let compiled = if ignore_case {
103                    StringMatcher::regex_ignore_case(v)
104                } else {
105                    StringMatcher::regex(v)
106                };
107                compiled
108                    .map(boxed)
109                    .map_err(|e| MatcherError::InvalidPattern {
110                        pattern: v.clone(),
111                        source: e.to_string(),
112                    })
113            }
114        }
115    }
116
117    /// Does this pattern turn case-insensitivity off with an inline flag?
118    ///
119    /// `ignore_case` is implemented by asking the regex engine for a
120    /// case-insensitive match, and an inline `(?-i)` overrides that — measured
121    /// both ways on 2026-08-18: `(?i)(?-i)admin` and
122    /// `RegexBuilder::case_insensitive(true)` on `(?-i)admin` both fail to
123    /// match `ADMIN`. That is correct regex semantics and not something a
124    /// different construction can fix, so the combination is rejected instead.
125    ///
126    /// Scans flag groups — `(?` followed by flag letters, then `)` or `:` — and
127    /// reports any that clear `i`. A group with no `-` cannot clear anything,
128    /// and `(?i-s)` clears `s`, not `i`.
129    /// Enforce the pattern length limit for this variant.
130    ///
131    /// Regexes get the tighter [`MAX_REGEX_PATTERN_LENGTH`](crate::MAX_REGEX_PATTERN_LENGTH)
132    /// because compiled program size, not pattern length, drives their cost.
133    fn check_length(&self) -> Result<(), MatcherError> {
134        let (len, max) = match self {
135            Self::Regex(v) => (v.len(), crate::MAX_REGEX_PATTERN_LENGTH),
136            Self::Exact(v) | Self::Prefix(v) | Self::Suffix(v) | Self::Contains(v) => {
137                (v.len(), crate::MAX_PATTERN_LENGTH)
138            }
139        };
140        if len > max {
141            return Err(MatcherError::PatternTooLong { len, max });
142        }
143        Ok(())
144    }
145
146    /// Compile this spec into a [`Predicate`] with the given [`DataInput`].
147    ///
148    /// Equivalent to `Predicate::Single(SinglePredicate::new(input, self.to_input_matcher()?))`.
149    ///
150    /// # Errors
151    ///
152    /// Returns [`MatcherError::InvalidPattern`] if the regex is invalid.
153    pub fn to_predicate<Ctx: 'static>(
154        &self,
155        input: Box<dyn DataInput<Ctx>>,
156    ) -> Result<Predicate<Ctx>, MatcherError> {
157        let matcher = self.to_input_matcher()?;
158        Ok(Predicate::Single(SinglePredicate::new(input, matcher)))
159    }
160}
161
162impl fmt::Display for StringMatchSpec {
163    fn fmt(&self, f: &mut fmt::Formatter<'_>) -> fmt::Result {
164        match self {
165            Self::Exact(v) => write!(f, "Exact(\"{v}\")"),
166            Self::Prefix(v) => write!(f, "Prefix(\"{v}\")"),
167            Self::Suffix(v) => write!(f, "Suffix(\"{v}\")"),
168            Self::Contains(v) => write!(f, "Contains(\"{v}\")"),
169            Self::Regex(v) => write!(f, "Regex(\"{v}\")"),
170        }
171    }
172}

334/// Does this pattern clear the `i` flag with an inline group?
335///
336/// `ignore_case` asks the regex engine for a case-insensitive match, and an
337/// inline `(?-i)` overrides that. Measured both ways on 2026-08-18:
338/// `(?i)(?-i)admin` and `RegexBuilder::case_insensitive(true)` applied to
339/// `(?-i)admin` both fail to match `ADMIN`. That is correct regex semantics, so
340/// no choice of construction fixes it — the combination has to be rejected.
341///
342/// Scans flag groups: `(?` followed by flag letters, terminated by `)` or `:`.
343/// A group clears `i` only if an `i` appears after a `-` within it, so `(?i-s)`
344/// is fine and `(?-si)` is not. Escaped `\(` is not a group.
345fn disables_case_insensitivity(pattern: &str) -> bool {
346    let bytes = pattern.as_bytes();
347    let mut i = 0;
348
349    while i + 1 < bytes.len() {
350        if bytes[i] == b'\\' {
351            i += 2;
352            continue;
353        }
354        if bytes[i] != b'(' || bytes[i + 1] != b'?' {
355            i += 1;
356            continue;
357        }
358
359        // Walk the flag letters. `-` switches from setting to clearing.
360        let mut j = i + 2;
361        let mut clearing = false;
362        while j < bytes.len() {
363            match bytes[j] {
364                b'-' => clearing = true,
365                b'i' if clearing => return true,
366                b'i' | b'm' | b's' | b'u' | b'x' | b'U' => {}
367                // `)` and `:` end a flag group; anything else means this was
368                // some other `(?` construct — a named group, a lookaround.
369                _ => break,
370            }
371            j += 1;
372        }
373        i = j.max(i + 2);
374    }
375
376    false
377}

379#[cfg(test)]
380mod ignore_case_tests {
381    use super::*;
382    use crate::MatchingData;
383
384    fn matches(spec: &StringMatchSpec, ignore_case: bool, input: &str) -> bool {
385        spec.to_input_matcher_with_case(ignore_case)
386            .unwrap()
387            .matches(&MatchingData::String(input.into()))
388    }
389
390    /// The flag was read from the proto and discarded, so a rule that read
391    /// case-insensitive matched case-sensitively.
392    #[test]
393    fn ignore_case_reaches_every_literal_variant() {
394        for spec in [
395            StringMatchSpec::Exact("admin".into()),
396            StringMatchSpec::Prefix("admin".into()),
397            StringMatchSpec::Suffix("admin".into()),
398            StringMatchSpec::Contains("admin".into()),
399        ] {
400            assert!(matches(&spec, true, "ADMIN"), "{spec:?} with ignore_case");
401            assert!(!matches(&spec, false, "ADMIN"), "{spec:?} without");
402        }
403    }
404
405    #[test]
406    fn ignore_case_reaches_regex() {
407        let spec = StringMatchSpec::Regex("^admin$".into());
408        assert!(matches(&spec, true, "ADMIN"));
409        assert!(!matches(&spec, false, "ADMIN"));
410    }
411
412    /// An inline `(?-i)` beats the engine's case-insensitive setting, whichever
413    /// way that setting is applied. So the pair is refused rather than silently
414    /// producing a rule that reads one way and behaves another.
415    #[test]
416    fn ignore_case_with_an_inline_off_switch_is_rejected() {
417        for pattern in ["(?-i)admin", "a(?-i)dmin", "(?-si)admin", "(?s-i)admin"] {
418            let spec = StringMatchSpec::Regex(pattern.into());
419            let err = spec.to_input_matcher_with_case(true).unwrap_err();
420            assert!(
421                matches!(err, MatcherError::InvalidPattern { .. }),
422                "{pattern} should be rejected, got {err:?}"
423            );
424            // Without ignore_case there is nothing to contradict.
425            assert!(spec.to_input_matcher_with_case(false).is_ok(), "{pattern}");
426        }
427    }
428
429    /// The guard must not reject patterns that clear some *other* flag, or that
430    /// merely contain the characters somewhere harmless.
431    #[test]
432    fn the_inline_flag_guard_does_not_overreach() {
433        for pattern in [
434            "(?i-s)admin",     // clears s, not i
435            "(?s)admin",       // sets s
436            r"\(?-i\)admin",   // escaped, not a group
437            "admin-i",         // no group at all
438            "(?:admin)",       // non-capturing group
439            "(?P<name>admin)", // named group
440        ] {
441            let spec = StringMatchSpec::Regex(pattern.into());
442            assert!(
443                spec.to_input_matcher_with_case(true).is_ok(),
444                "{pattern} should be accepted"
445            );
446        }
