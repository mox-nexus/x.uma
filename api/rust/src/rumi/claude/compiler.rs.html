13/// A catch-all predicate that matches any `HookContext`.
14fn catch_all() -> Predicate<HookContext> {
15    Predicate::Single(SinglePredicate::new(
16        Box::new(EventTypeInput),
17        Box::new(PrefixMatcher::new("")), // matches any event string
18    ))
19}

21/// Extension trait for compiling `HookMatch` to rumi `Matcher`.
22pub trait HookMatchExt {
23    /// Compile this `HookMatch` into a rumi `Matcher`.
24    ///
25    /// The resulting matcher operates on [`HookContext`] and returns
26    /// the provided action when all conditions match.
27    ///
28    /// # Errors
29    ///
30    /// Returns [`MatcherError`] if any regex pattern is invalid.
31    fn compile<A: Clone + Send + Sync + 'static>(
32        &self,
33        action: A,
34    ) -> Result<Matcher<HookContext, A>, MatcherError>;
35
36    /// Compile this `HookMatch` into a `Predicate` (without action).
37    ///
38    /// # Errors
39    ///
40    /// Returns [`MatcherError`] if any regex pattern is invalid.
41    fn to_predicate(&self) -> Result<Predicate<HookContext>, MatcherError>;
42}
43
44impl HookMatchExt for HookMatch {
45    fn compile<A: Clone + Send + Sync + 'static>(
46        &self,
47        action: A,
48    ) -> Result<Matcher<HookContext, A>, MatcherError> {
49        let predicate = self.to_predicate()?;
50        Ok(Matcher::from_predicate(predicate, action, None))
51    }
52
53    fn to_predicate(&self) -> Result<Predicate<HookContext>, MatcherError> {
54        let mut predicates: Vec<Predicate<HookContext>> = Vec::new();
55
56        // Event matching (exact only — event names are a known enum)
57        if let Some(event) = &self.event {
58            predicates.push(compile_event_match(*event));
59        }
60
61        // Tool name matching
62        if let Some(tool_match) = &self.tool_name {
63            predicates.push(tool_match.to_predicate(Box::new(ToolNameInput))?);
64        }
65
66        // Argument matching (all ANDed)
67        if let Some(arguments) = &self.arguments {
68            for arg_match in arguments {
69                predicates.push(compile_argument_match(arg_match)?);
70            }
71        }
72
73        // Session id matching
74        if let Some(session_match) = &self.session_id {
75            predicates.push(session_match.to_predicate(Box::new(SessionIdInput))?);
76        }
77
78        // CWD matching
79        if let Some(cwd_match) = &self.cwd {
80            predicates.push(cwd_match.to_predicate(Box::new(CwdInput))?);
81        }
82
83        // Git branch matching
84        if let Some(branch_match) = &self.git_branch {
85            predicates.push(branch_match.to_predicate(Box::new(GitBranchInput))?);
86        }
87
88        // An empty conjunction is vacuously true, so `from_all` would hand back
89        // `catch_all()` — a rule the operator believes is scoped, silently
90        // matching every context. This is the only moment the mistake is
91        // visible: after substitution the predicate is `PrefixMatcher("")` and
92        // is indistinguishable from a deliberate catch-all, which is why
93        // `Matcher::validate` cannot catch it and never could.
94        if predicates.is_empty() && !self.match_all {
95            return Err(MatcherError::InvalidConfig {
96                source: "HookMatch has no conditions, so it matches every hook \
97                         invocation. Set `match_all: true` if that is intended, \
98                         or constrain the rule."
99                    .into(),
100            });
101        }
102
103        Ok(Predicate::from_all(predicates, catch_all()))
104    }
105}

107/// Compile an event match to a predicate (always exact).
108fn compile_event_match(event: HookEvent) -> Predicate<HookContext> {
109    Predicate::Single(SinglePredicate::new(
110        Box::new(EventTypeInput),
111        Box::new(ExactMatcher::new(event.as_str())),
112    ))
113}

115/// Compile an argument match to a predicate.
116fn compile_argument_match(
117    arg_match: &ArgumentMatch,
118) -> Result<Predicate<HookContext>, MatcherError> {
119    arg_match
120        .value
121        .to_predicate(Box::new(ToolArgInput::new(&arg_match.name)?))

124/// Compile multiple `HookMatch` entries into a single `Matcher`.
125///
126/// Multiple matches are `ORed` together: the first matching rule wins.
127///
128/// # Errors
129///
130/// Returns [`MatcherError`] if any regex pattern is invalid, if the rule list
131/// is empty, or if any rule has no conditions without `match_all`. The last two
132/// are catch-alls, and this is the path that gates agent tool calls.
133pub fn compile_hook_matches<A: Clone + Send + Sync + 'static>(
134    matches: &[HookMatch],
135    action: A,
136    on_no_match: Option<A>,
137) -> Result<Matcher<HookContext, A>, MatcherError> {
138    // An empty disjunction is vacuously true in the same way an empty
139    // conjunction is — and xDS says the opposite, explicitly: "if no matcher
140    // above matched ... the match will be considered unsuccessful." The config
141    // path already honours that. This convenience layer contradicted it.
142    //
143    // Not fixed by copying the loader, because here `on_no_match` is an
144    // argument rather than config: `(&[], "allow", Some("deny"))` and
145    // `(&[], "deny", Some("allow"))` are opposite outcomes from the same empty
146    // input. The usual route to an empty list is not writing one — it is a
147    // config file that failed to load, or a filter that removed every rule.
148    if matches.is_empty() {
149        return Err(MatcherError::InvalidConfig {
150            source: "no hook rules, which would match every hook invocation. \
151                     Use `compile_catch_all` if that is intended."
152                .into(),
153        });
154    }
155
156    let predicates: Vec<Predicate<HookContext>> = matches
157        .iter()
158        .map(HookMatchExt::to_predicate)
159        .collect::<Result<_, _>>()?;
160
161    let or_pred = Predicate::from_any(predicates, catch_all());
162    let matcher = Matcher::from_predicate(or_pred, action, on_no_match);
163
164    // See the note in rumi-http's compiler: a compiled matcher must carry the
165    // same guarantees as a loaded one, and this is the path that gates agent
166    // tool calls.
167    matcher.validate()?;
168    Ok(matcher)
169}

171/// Build a matcher that matches every hook invocation.
172///
173/// The explicit form of what [`compile_hook_matches`] now refuses to do by
174/// accident. Nothing here is unsafe — a catch-all is a legitimate rule — but it
175/// has to be asked for, and it is greppable when someone later asks why an
176/// allowlist admits everything.
177///
178/// # Errors
179///
180/// Returns [`MatcherError`] only if the resulting matcher fails validation,
181/// which it cannot at this depth. The `Result` is kept for symmetry.
182pub fn compile_catch_all<A: Clone + Send + Sync + 'static>(
183    action: A,
184) -> Result<Matcher<HookContext, A>, MatcherError> {
185    let matcher = Matcher::from_predicate(catch_all(), action, None);
186    matcher.validate()?;
187    Ok(matcher)
188}

194/// Result of tracing a `HookMatch` against a `HookContext`.
195///
196/// Shows which fields matched and which didn't, with expected vs actual values.
197#[derive(Debug)]
198pub struct HookMatchTrace {
199    /// Whether the overall match succeeded (all steps must match).
200    pub matched: bool,
201    /// Per-field trace steps.
202    pub steps: Vec<TraceStep>,
203}
204
205/// One field's evaluation result in a trace.
206#[derive(Debug)]
207pub struct TraceStep {
208    /// Field name (e.g., "event", "`tool_name`", "argument\[command\]").
209    pub field: String,
210    /// What was expected (e.g., `Exact("Bash")`).
211    pub expected: String,
212    /// What was found in the context.
213    pub actual: String,
214    /// Whether this individual step matched.
215    pub matched: bool,
216}
217
218impl HookMatch {
219    /// Trace this match against a context, reporting per-field results.
220    ///
221    /// Use this to debug "why was this hook triggered?" or "why didn't it match?"
222    #[must_use]
223    pub fn trace(&self, ctx: &HookContext) -> HookMatchTrace {
224        let mut steps = Vec::new();
225
226        if let Some(event) = &self.event {
227            let actual = ctx.event().as_str().to_string();
228            let expected_str = event.as_str().to_string();
229            let matched = actual == expected_str;
230            steps.push(TraceStep {
231                field: "event".into(),
232                expected: expected_str,
233                actual,
234                matched,
235            });
236        }
237
238        if let Some(tool_match) = &self.tool_name {
239            let actual = ctx.tool_name().to_string();
240            let matched = trace_string_match(tool_match, &actual);
241            steps.push(TraceStep {
242                field: "tool_name".into(),
243                expected: tool_match.to_string(),
244                actual,
245                matched,
246            });
247        }
248
249        if let Some(arguments) = &self.arguments {
250            for arg_match in arguments {
251                let actual = ctx
252                    .argument(&arg_match.name)
253                    .unwrap_or("<missing>")
254                    .to_string();
255                let matched = ctx
256                    .argument(&arg_match.name)
257                    .is_some_and(|v| trace_string_match(&arg_match.value, v));
258                steps.push(TraceStep {
259                    field: format!("argument[{}]", arg_match.name),
260                    expected: arg_match.value.to_string(),
261                    actual,
262                    matched,
263                });
264            }
265        }
266
267        if let Some(cwd_match) = &self.cwd {
268            let actual = ctx.cwd().to_string();
269            let matched = trace_string_match(cwd_match, &actual);
270            steps.push(TraceStep {
271                field: "cwd".into(),
272                expected: cwd_match.to_string(),
273                actual,
274                matched,
275            });
276        }
277
278        if let Some(branch_match) = &self.git_branch {
279            let actual = ctx.git_branch().unwrap_or("<none>").to_string();
280            let matched = ctx
281                .git_branch()
282                .is_some_and(|v| trace_string_match(branch_match, v));
283            steps.push(TraceStep {
284                field: "git_branch".into(),
285                expected: branch_match.to_string(),
286                actual,
287                matched,
288            });
289        }
290
291        let overall = steps.iter().all(|s| s.matched);
292        HookMatchTrace {
293            matched: overall,
294            steps,
295        }
296    }
297}

299/// Evaluate a `StringMatch` against a concrete value (for trace only).
300///
301/// Compiles the `StringMatch` to an `InputMatcher` and checks against
302/// `MatchingData::String`. Returns false if the regex is invalid.
303fn trace_string_match(string_match: &StringMatch, value: &str) -> bool {
304    string_match
305        .to_input_matcher()
306        .is_ok_and(|m| m.matches(&MatchingData::String(value.to_string())))
307}
