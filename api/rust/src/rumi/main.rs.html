1//! rumi CLI — driving adapter for the rumi matcher engine.
2//!
3//! Supports three domains:
4//! - **test** (default) — key-value context pairs
5//! - **http** — HTTP method, path, headers, query params
6//! - **claude** — Claude Code hook events
7//!
8//! Subcommands:
9//! - `run [http|claude] <config> [flags]` — run config against context
10//! - `check [http|claude] <config>` — validate config loads without errors
11//! - `info [http|claude]` — print registered type URLs
12

24fn main() {
25    let args: Vec<String> = std::env::args().collect();
26
27    if args.len() < 2 {
28        print_usage();
29        process::exit(1);
30    }
31
32    let result = match args[1].as_str() {
33        "run" => cmd_run(&args[2..]),
34        "check" => cmd_check(&args[2..]),
35        "info" => cmd_info(&args[2..]),
36        "--skill" => cmd_skill(&args[2..]),
37        "--help" | "-h" | "help" => {
38            print_usage();
39            Ok(())
40        }
41        other => {
42            eprintln!("error: unknown command \"{other}\"");
43            print_usage();
44            process::exit(1);
45        }
46    };
47
48    if let Err(e) = result {
49        eprintln!("error: {e}");
50        process::exit(1);
51    }
52}

58#[derive(Debug, Clone, Copy, PartialEq, Eq)]
59enum Domain {
60    Test,
61    Http,
62    Claude,
63}

65/// Detect the domain from the first argument. Returns `(domain, remaining_args)`.
66fn detect_domain(args: &[String]) -> (Domain, &[String]) {
67    match args.first().map(String::as_str) {
68        Some("http") => (Domain::Http, &args[1..]),
69        Some("claude") => (Domain::Claude, &args[1..]),
70        _ => (Domain::Test, args),
71    }
72}

78fn cmd_run(args: &[String]) -> Result<(), String> {
79    let (domain, rest) = detect_domain(args);
80    match domain {
81        Domain::Test => cmd_run_test(rest),
82        Domain::Http => cmd_run_http(rest),
83        Domain::Claude => cmd_run_claude(rest),
84    }
85}

87fn cmd_check(args: &[String]) -> Result<(), String> {
88    let (domain, rest) = detect_domain(args);
89
90    if rest.is_empty() {
91        return Err("check requires a config file path".into());
92    }
93
94    let config_path = &rest[0];
95    let config = load_config(config_path)?;
96
97    match domain {
98        Domain::Test => {
99            let registry = build_test_registry();
100            registry
101                .load_matcher(config)
102                .map_err(|e| format!("config invalid: {e}"))?;
103        }
104        Domain::Http => {
105            let registry = build_http_registry();
106            registry
107                .load_matcher(config)
108                .map_err(|e| format!("config invalid: {e}"))?;
109        }
110        Domain::Claude => {
111            let registry = build_claude_registry();
112            registry
113                .load_matcher(config)
114                .map_err(|e| format!("config invalid: {e}"))?;
115        }
116    }
117
118    println!("Config valid");
119    Ok(())
120}

123/// `rumi --skill [-r NAME]` — emit agent-facing documentation.
124///
125/// Type URL tables come from the live registries, so the document cannot
126/// describe an input that is not actually registered.
127fn cmd_skill(args: &[String]) -> Result<(), String> {
128    if let Some(pos) = args.iter().position(|a| a == "-r" || a == "--reference") {
129        let name = args.get(pos + 1).ok_or_else(|| {
130            format!(
131                "--skill -r requires a reference name (one of: {})",
132                skill::REFERENCE_NAMES.join(", ")
133            )
134        })?;
135        let body = skill::reference(name).ok_or_else(|| {
136            format!(
137                "unknown reference \"{name}\" (one of: {})",
138                skill::REFERENCE_NAMES.join(", ")
139            )
140        })?;
141        print!("{body}");
142        return Ok(());
143    }
144
145    let test = build_test_registry();
146    let http = build_http_registry();
147    let claude = build_claude_registry();
148
149    print!(
150        "{}",
151        skill::skill(
152            &test.input_type_urls(),
153            &test.matcher_type_urls(),
154            &http.input_type_urls(),
155            &claude.input_type_urls(),
156        )
157    );
158    Ok(())
159}

161fn cmd_info(args: &[String]) -> Result<(), String> {
162    let (domain, _) = detect_domain(args);
163
164    match domain {
165        Domain::Test => print_registry_info(&build_test_registry()),
166        Domain::Http => print_registry_info(&build_http_registry()),
167        Domain::Claude => print_registry_info(&build_claude_registry()),
168    }
169
170    Ok(())
171}

173fn print_registry_info<Ctx: 'static>(registry: &rumi::Registry<Ctx>) {
174    println!("Registered inputs:");
175    for url in registry.input_type_urls() {
176        println!("  {url}");
177    }
178
179    println!("\nRegistered matchers:");
180    for url in registry.matcher_type_urls() {
181        println!("  {url}");
182    }
183}

189/// True if `--trace` appears anywhere in the args.
190fn wants_trace(args: &[String]) -> bool {
191    args.iter().any(|a| a == "--trace")
192}
193
194/// Strip `--trace` so domain arg parsers never see it.
195fn without_trace(args: &[String]) -> Vec<String> {
196    args.iter().filter(|a| *a != "--trace").cloned().collect()
197}

199/// Evaluate, printing either the action or the full trace.
200fn report<Ctx, A: std::fmt::Display + Clone + Send + Sync + 'static>(
201    matcher: &rumi::Matcher<Ctx, A>,
202    ctx: &Ctx,
203    trace: bool,
204) {
205    if trace {
206        trace_output::print(&matcher.evaluate_with_trace(ctx));
207    } else {
208        match matcher.evaluate(ctx) {
209            Some(action) => println!("{action}"),
210            None => println!("(no match)"),
211        }
212    }
213}

215fn cmd_run_test(args: &[String]) -> Result<(), String> {
216    if args.is_empty() {
217        return Err("run requires a config file path".into());
218    }
219
220    let trace = wants_trace(args);
221    let args = without_trace(args);
222    let config_path = &args[0];
223    let context = parse_context_pairs(&args[1..])?;
224
225    let config = load_config(config_path)?;
226    let registry = build_test_registry();
227    let matcher = registry
228        .load_matcher(config)
229        .map_err(|e| format!("config load failed: {e}"))?;
230
231    let ctx = build_test_context(&context);
232    report(&matcher, &ctx, trace);
233
234    Ok(())
235}

313    let mut ctx = TestContext::new();
314    for (k, v) in pairs {
315        ctx = ctx.with(k, v);
316    }
317    ctx

320fn build_http_context(args: &HttpArgs) -> HttpRequest {
321    let mut builder = HttpRequest::builder().method(&args.method).path(&args.path);
322    for (k, v) in &args.headers {
323        builder = builder.header(k, v);
324    }
325    for (k, v) in &args.query_params {
326        builder = builder.query_param(k, v);
327    }
328    builder.build()
329}

331fn build_claude_context(args: &ClaudeArgs) -> Result<HookContext, String> {
332    let mut ctx = match args.event.as_str() {
333        "PreToolUse" => HookContext::pre_tool_use(&args.tool),
334        "PostToolUse" => HookContext::post_tool_use(&args.tool),
335        "Stop" => HookContext::stop(),
336        "SubagentStop" => HookContext::subagent_stop(),
337        "UserPromptSubmit" => HookContext::user_prompt_submit(),
338        "SessionStart" => HookContext::session_start(),
339        "SessionEnd" => HookContext::session_end(),
340        "PreCompact" => HookContext::pre_compact(),
341        "Notification" => HookContext::notification(),
342        other => {
343            return Err(format!(
344                "unknown event \"{other}\". Valid events: PreToolUse, PostToolUse, Stop, \
345                 SubagentStop, UserPromptSubmit, SessionStart, SessionEnd, PreCompact, Notification"
346            ))
347        }
348    };
349
350    for (k, v) in &args.arguments {
351        ctx = ctx.with_arg(k, v);
352    }
353    if let Some(ref cwd) = args.cwd {
354        ctx = ctx.with_cwd(cwd);
355    }
356    if let Some(ref branch) = args.branch {
357        ctx = ctx.with_git_branch(branch);
358    }
359    if let Some(ref session) = args.session_id {
360        ctx = ctx.with_session_id(session);
361    }
362
363    Ok(ctx)
364}

370fn load_config(path: &str) -> Result<MatcherConfig<String>, String> {
371    let content =
372        std::fs::read_to_string(path).map_err(|e| format!("failed to read \"{path}\": {e}"))?;
373
374    let is_json = std::path::Path::new(path)
375        .extension()
376        .is_some_and(|ext| ext.eq_ignore_ascii_case("json"));
377
378    if is_json {
379        serde_json::from_str(&content).map_err(|e| format!("JSON parse error: {e}"))
380    } else {
381        serde_yaml::from_str(&content).map_err(|e| format!("YAML parse error: {e}"))
382    }
383}

389/// Parsed HTTP arguments.

391struct HttpArgs {
392    method: String,
393    path: String,
394    headers: Vec<(String, String)>,
395    query_params: Vec<(String, String)>,
396}

398/// Parsed Claude arguments.
399#[derive(Debug, Default)]
400struct ClaudeArgs {
401    event: String,
402    tool: String,
403    arguments: Vec<(String, String)>,
404    cwd: Option<String>,
405    branch: Option<String>,
406    session_id: Option<String>,
407}
408
409fn parse_http_args(args: &[String]) -> Result<HttpArgs, String> {
410    let mut result = HttpArgs::default();
411    let mut i = 0;
412
413    while i < args.len() {
414        match args[i].as_str() {
415            "--method" => {
416                i += 1;
417                result.method = next_value(args, i, "--method")?;
418                i += 1;
419            }
420            "--path" => {
421                i += 1;
422                result.path = next_value(args, i, "--path")?;
423                i += 1;
424            }
425            "--header" => {
426                i += 1;
427                let pair = next_value(args, i, "--header")?;
428                let (k, v) = split_kv(&pair, "--header")?;
429                result.headers.push((k, v));
430                i += 1;
431            }
432            "--query" => {
433                i += 1;
434                let pair = next_value(args, i, "--query")?;
435                let (k, v) = split_kv(&pair, "--query")?;
436                result.query_params.push((k, v));
437                i += 1;
438            }
439            other => return Err(format!("unexpected argument \"{other}\"")),
440        }
441    }
442
443    if result.method.is_empty() {
444        return Err("--method is required for http domain".into());
445    }
446    if result.path.is_empty() {
447        return Err("--path is required for http domain".into());
448    }
449
450    Ok(result)
451}

453fn parse_claude_args(args: &[String]) -> Result<ClaudeArgs, String> {
454    let mut result = ClaudeArgs::default();
455    let mut i = 0;
456
457    while i < args.len() {
458        match args[i].as_str() {
459            "--event" => {
460                i += 1;
461                result.event = next_value(args, i, "--event")?;
462                i += 1;
463            }
464            "--tool" => {
465                i += 1;
466                result.tool = next_value(args, i, "--tool")?;
467                i += 1;
468            }
469            "--arg" => {
470                i += 1;
471                let pair = next_value(args, i, "--arg")?;
472                let (k, v) = split_kv(&pair, "--arg")?;
473                result.arguments.push((k, v));
474                i += 1;
475            }
476            "--cwd" => {
477                i += 1;
478                result.cwd = Some(next_value(args, i, "--cwd")?);
479                i += 1;
480            }
481            "--branch" => {
482                i += 1;
483                result.branch = Some(next_value(args, i, "--branch")?);
484                i += 1;
485            }
486            "--session" => {
487                i += 1;
488                result.session_id = Some(next_value(args, i, "--session")?);
489                i += 1;
490            }
491            other => return Err(format!("unexpected argument \"{other}\"")),
492        }
493    }
494
495    if result.event.is_empty() {
496        return Err("--event is required for claude domain".into());
497    }
498
499    Ok(result)
500}

502/// Parse `--context key=value...` pairs (test domain).
503fn parse_context_pairs(args: &[String]) -> Result<HashMap<String, String>, String> {
504    let mut map = HashMap::new();
505    let mut i = 0;
506
507    while i < args.len() {
508        if args[i] == "--context" {
509            i += 1;
510            while i < args.len() && !args[i].starts_with("--") {
511                let pair = &args[i];
512                let (key, value) = pair.split_once('=').ok_or_else(|| {
513                    format!("invalid context pair \"{pair}\", expected key=value")
514                })?;
515                map.insert(key.to_owned(), value.to_owned());
516                i += 1;
517            }
518        } else {
519            return Err(format!("unexpected argument \"{}\"", args[i]));
520        }
521    }
522
523    Ok(map)
524}

526/// Get the next argument value, erroring if missing.
527fn next_value(args: &[String], i: usize, flag: &str) -> Result<String, String> {
528    args.get(i)
529        .cloned()
530        .ok_or_else(|| format!("{flag} requires a value"))
531}
532
533/// Split a `key=value` pair.
534fn split_kv(pair: &str, flag: &str) -> Result<(String, String), String> {
535    pair.split_once('=')
536        .map(|(k, v)| (k.to_owned(), v.to_owned()))
537        .ok_or_else(|| format!("invalid {flag} pair \"{pair}\", expected key=value"))
538}

858    // ─── Skill accuracy ──────────────────────────────────────────────────
859
860    /// Every config key named in `skill::CONFIG_KEYS` must be the key the real
861    /// loader accepts.
862    ///
863    /// This test exists because the hand-written half of `skill.rs` claimed
864    /// `ArgumentInput` reads `config.key` when the struct field is `name`. The
865    /// generated half of that file cannot drift; this half could, and did,
866    /// within hours of being written. A doc claim that CI does not check is not
867    /// a doc claim, it is a guess.
868    #[test]
869    fn skill_hints_name_real_config_keys() {
870        for (type_url, key) in skill::CONFIG_KEYS {
871            let yaml = format!(
872                r#"
873matchers:
874  - predicate:
875      type: single
876      input: {{ type_url: "{type_url}", config: {{ {key}: "x" }} }}
877      value_match: {{ Exact: "y" }}
878    on_match: {{ type: action, action: "a" }}
879"#
880            );
881            let config: MatcherConfig<String> =
882                serde_yaml::from_str(&yaml).unwrap_or_else(|e| panic!("{type_url}: {e}"));
883
884            let loaded = if type_url.starts_with("xuma.http.") {
885                build_http_registry().load_matcher(config).map(|_| ())
886            } else if type_url.starts_with("xuma.claude.") {
887                build_claude_registry().load_matcher(config).map(|_| ())
888            } else {
889                build_test_registry().load_matcher(config).map(|_| ())
890            };
891
892            assert!(
893                loaded.is_ok(),
894                "skill.rs documents `{type_url}` as taking config.{key}, but the \
895                 loader rejected it: {:?}",
896                loaded.err()
897            );
898        }
899    }

903    fn skill_hint_test_would_catch_a_wrong_key() {
904        let yaml = r#"
