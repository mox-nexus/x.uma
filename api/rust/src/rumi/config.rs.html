1//! Config types for generic matcher construction.
2//!
3//! These mirror the runtime matcher types and are what
4//! [`Registry::load_matcher()`] consumes. They are **built, not parsed**: the
5//! authored config format is canonical protojson, read by `rumi-proto` into
6//! xDS proto types and converted here.
7//!
8//! None of the tree-shape types below derives `Deserialize`, and that is the
9//! enforcement rather than a style choice. While they did, a second, terser
10//! JSON dialect remained authorable in Rust — one no fixture used, no other
11//! implementation could read, and nothing documented. Removing the impl makes
12//! writing that dialect a compile error instead of a convention.
13//!
14//! The payload types keep theirs: a `TypedConfig.config` body is deserialized
15//! as the registered factory's `Config` associated type, which is the
16//! extension seam working as designed.
17//!
18//! # Relationship to runtime types
19//!
20//! | Config type | Runtime type | Loader method |
21//! |-------------|-------------|---------------|
22//! | [`MatcherConfig`] | [`Matcher`](crate::Matcher) | `Registry::load_matcher()` |
23//! | [`FieldMatcherConfig`] | [`FieldMatcher`](crate::FieldMatcher) | `Registry::load_field_matcher()` |
24//! | [`PredicateConfig`] | [`Predicate`](crate::Predicate) | `Registry::load_predicate()` |
25//! | [`SinglePredicateConfig`] | [`SinglePredicate`](crate::SinglePredicate) | `Registry::load_single()` |
26//! | [`ValueMatchConfig`] | `Box<dyn InputMatcher>` | built-in or via registry factory |
27//! | [`OnMatchConfig`] | [`OnMatch`](crate::OnMatch) | `Registry::load_on_match()` |
28//! | [`TypedConfig`] | `Box<dyn DataInput<Ctx>>` or `Box<dyn InputMatcher>` | via registry factory |

33/// Configuration for a [`Matcher`](crate::Matcher).
34///
35/// Built by a protojson reader, then loaded into a runtime `Matcher` via
36/// [`Registry::load_matcher()`](crate::Registry::load_matcher).
37#[derive(Debug, Clone)]
38pub struct MatcherConfig<A> {
39    /// A list of field matchers, or a lookup tree. Never both.
40    pub kind: MatcherKindConfig<A>,
41
42    /// Fallback when nothing matched.
43    pub on_no_match: Option<OnMatchConfig<A>>,
44}
45
46impl<A> MatcherConfig<A> {
47    /// A list matcher, which is what most configs are.
48    #[must_use]
49    pub fn list(matchers: Vec<FieldMatcherConfig<A>>) -> Self {
50        Self {
51            kind: MatcherKindConfig::List(matchers),
52            on_no_match: None,
53        }
54    }
55
56    /// Set the fallback.
57    #[must_use]
58    pub fn with_fallback(mut self, on_no_match: Option<OnMatchConfig<A>>) -> Self {
59        self.on_no_match = on_no_match;
60        self
61    }
62}

64/// Config mirror of [`MatcherKind`](crate::MatcherKind) — xDS `oneof matcher_type`.
65#[derive(Debug, Clone)]
66pub enum MatcherKindConfig<A> {
67    /// Field matchers evaluated in order.
68    List(Vec<FieldMatcherConfig<A>>),
69    /// A single map lookup.
70    Tree(MatcherTreeConfig<A>),
71}
72
73/// Config for a [`MatcherTree`](crate::MatcherTree).
74///
75/// Carries no fallback: the proto `MatcherTree` has no such field and the
76/// enclosing `Matcher` owns it. See `DECISIONS.md` D-044.
77#[derive(Debug, Clone)]
78pub struct MatcherTreeConfig<A> {
79    /// The input producing the lookup key, resolved via the registry.
80    pub input: TypedConfig,
81    /// Which lookup rule applies, and the entries it applies to.
82    pub tree: TreeTypeConfig<A>,
83}
84
85/// Config mirror of xDS `MatcherTree.tree_type`.
86#[derive(Debug, Clone)]
87pub enum TreeTypeConfig<A> {
88    /// O(1) exact key lookup.
89    ExactMatchMap(Vec<(String, OnMatchConfig<A>)>),
90    /// O(k) longest-prefix lookup.
91    PrefixMatchMap(Vec<(String, OnMatchConfig<A>)>),
92}

94/// Configuration for a [`FieldMatcher`](crate::FieldMatcher).
95#[derive(Debug, Clone)]
96pub struct FieldMatcherConfig<A> {
97    /// The predicate that gates this field matcher.
98    pub predicate: PredicateConfig,
99
100    /// What to do when the predicate matches.
101    pub on_match: OnMatchConfig<A>,
102}

104/// Configuration for a [`Predicate`](crate::Predicate).
105///
106/// Mirrors xDS `Matcher.MatcherList.Predicate`, whose `oneof match_type` is
107/// these four arms.
108#[derive(Debug, Clone)]
109pub enum PredicateConfig {
110    /// A single predicate: input + value match.
111    Single(SinglePredicateConfig),
112
113    /// All predicates must match (logical AND).
114    And {
115        /// Child predicates (all must match).
116        predicates: Vec<PredicateConfig>,
117    },
118
119    /// Any predicate must match (logical OR).
120    Or {
121        /// Child predicates (any must match).
122        predicates: Vec<PredicateConfig>,
123    },
124
125    /// Inverts the inner predicate (logical NOT).
126    Not {
127        /// The predicate to negate.
128        predicate: Box<PredicateConfig>,
129    },
130}
131
132/// How to match the extracted value in a [`SinglePredicateConfig`].
133///
134/// Mirrors Envoy's `oneof matcher` in `SinglePredicate`:
135/// - `BuiltIn` — built-in string matching (Envoy: `StringMatcher value_match`)
136/// - `Custom` — custom matcher via registry (Envoy: `TypedExtensionConfig custom_match`)
137///
138/// The enum makes illegal states unrepresentable: exactly one variant is active.
139#[derive(Debug, Clone)]
140pub enum ValueMatchConfig {
141    /// Built-in string matching (exact, prefix, suffix, contains, regex).
142    BuiltIn {
143        /// The pattern and how to apply it.
144        spec: StringMatchSpec,
145        /// Case-insensitive matching — xDS `StringMatcher.ignore_case`.
146        ///
147        /// Carried here rather than inside `StringMatchSpec` because it is a
148        /// property of the comparison, not of the pattern. The proto path used
149        /// to drop it on the floor.
150        ignore_case: bool,
151    },
152    /// Custom matcher resolved via the registry's matcher factories.
153    Custom(TypedConfig),
154}

156/// Configuration for a [`SinglePredicate`](crate::SinglePredicate).
157///
158/// Combines a typed input reference (resolved via registry) with a value
159/// matcher. Mirrors xDS `SinglePredicate`, whose `oneof matcher` is either a
160/// built-in `StringMatcher` or a `TypedExtensionConfig`.
161///
162/// The exactly-one-of rule is enforced where the config is built: the proto
163/// `oneof` makes both-set unrepresentable, and `rumi-proto`'s converter
164/// rejects neither-set.
165#[derive(Debug, Clone)]
166pub struct SinglePredicateConfig {
167    /// The input to extract data from context.
168    /// Resolved at load time via the registry's `type_url` lookup.
169    pub input: TypedConfig,
170
171    /// How to match the extracted value.
172    pub matcher: ValueMatchConfig,
173}

175/// Reference to a registered type with its configuration.
176///
177/// Maps to xDS `TypedExtensionConfig`:
178/// - `type_url` identifies the registered type (input, matcher, or action)
179/// - `config` carries the type-specific configuration payload
180#[derive(Debug, Clone)]
181pub struct TypedConfig {
182    /// The type URL identifying the registered type.
183    /// Must match a `type_url` registered in the [`Registry`](crate::Registry).
184    pub type_url: String,
185
186    /// Type-specific configuration payload.
187    /// Deserialized as the `Config` associated type of the registered trait impl.
188    pub config: serde_json::Value,
189}
190
191/// Empty configuration for [`DataInput`](crate::DataInput) types that need no configuration.
192///
193/// Accepts any JSON value (`{}`, `null`, etc.) and ignores it.
194/// Use as the `Config` associated type in [`IntoDataInput`](crate::IntoDataInput)
195/// for inputs that are self-contained (no construction parameters).
196#[derive(Debug, Clone, Copy)]
197pub struct UnitConfig;
198
199impl<'de> Deserialize<'de> for UnitConfig {
200    fn deserialize<D: serde::Deserializer<'de>>(deserializer: D) -> Result<Self, D::Error> {
201        serde::de::IgnoredAny::deserialize(deserializer)?;
202        Ok(UnitConfig)
203    }
204}

206/// Configuration for [`OnMatch`](crate::OnMatch).
207///
208/// Either an action (leaf) or a nested matcher (tree).
209/// `OnMatch` exclusivity is enforced by the enum: action XOR matcher.
210#[derive(Debug, Clone)]
211pub enum OnMatchConfig<A> {
212    /// Return this action when the predicate matches.
213    Action {
214        /// The action value.
215        action: A,
216    },
217
218    /// Evaluate a nested matcher when the predicate matches.
219    Matcher {
220        /// The nested matcher configuration.
221        matcher: Box<MatcherConfig<A>>,
222    },
223}
